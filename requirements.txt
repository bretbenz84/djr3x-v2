# DJ-R3X v2 — Python dependencies
# Install inside the venv:  pip install -r requirements.txt
# Requires macOS Apple Silicon. Run setup_macos.sh first for system libs.

# ── Apple Silicon ML (mlx must come before mlx-whisper) ──────────────────────
# mlx floor raised to 0.32.0: with the Qwen3-TTS engine in-process, mlx 0.31.2
# hit a fatal native crash right after a generation completed (a Metal-side
# thread calling into Python with no thread state — PyThreadState_Get / trap 5,
# dev mac 2026-07-19, twice). 0.32.0 is the version the proven-stable POC ran.
mlx>=0.32.0
mlx-whisper>=0.4.0

# ── On-device TTS (Qwen3-TTS voice clone — --local-tts mode + ElevenLabs
#    fallback + impersonation). Pulls in transformers/tokenizers/mistral-common;
#    on Python <=3.12 mistral-common caps numpy<2.4, so installing this downgrades
#    numpy from 2.4.x to 2.3.x (verified compatible with numba/scipy/opencv here).
#    Model weights (~2.9 GB) are fetched by setup_assets.py, not pip. ──────────
mlx-audio[tts]==0.4.5

# ── Hugging Face (model download for whisper) ─────────────────────────────────
huggingface-hub>=0.23.0

# ── OpenAI (LLM, vision, Whisper API fallback) ────────────────────────────────
openai>=1.40.0

# ── ElevenLabs TTS ────────────────────────────────────────────────────────────
elevenlabs>=1.9.0

# ── Audio stream & VAD ────────────────────────────────────────────────────────
sounddevice>=0.4.7
pygame>=2.5.2
silero-vad>=5.1.2
numpy>=1.26.0
scipy>=1.13.0

# ── Wake word (OpenWakeWord + ONNX runtime) ───────────────────────────────────
openwakeword>=0.6.0
onnxruntime>=1.18.0

# ── Speaker identification ────────────────────────────────────────────────────
# Primary embedder: ECAPA-TDNN via SpeechBrain (VOICE_EMBEDDER="ecapa", 192-dim;
# model downloaded by setup_assets.py). Resemblyzer is the legacy fallback.
speechbrain>=1.0.0
resemblyzer>=0.1.1.dev0
# torch family is version-locked together AND to torchvision (added below for the
# place-recognition encoder). The <2.13 / <0.28 caps keep pip on the 2.12 line that
# RF-DETR + open_clip are validated against, so installing the encoder can't silently
# upgrade torch out from under the rest of the vision stack.
torch>=2.3.0,<2.13
torchaudio>=2.3.0,<2.13

# ── Vision — face detection & recognition ────────────────────────────────────
# Primary backend: InsightFace (SCRFD detector + ArcFace 512-dim recognizer) on
# ONNX Runtime (already above). Models downloaded by setup_assets.py.
insightface>=0.7.3
# Legacy fallback backend (FACE_BACKEND=dlib, or automatic if InsightFace fails
# to load). setup_macos.sh installs dlib separately with Apple Silicon libpng
# build flags.
dlib>=19.24.6
Pillow>=10.0.0

# ── Vision — pose estimation + local face expression telemetry ────────────────
# mediapipe installs opencv-contrib-python, which already provides cv2
mediapipe>=0.10.14

# ── Vision — local animal/object detection ────────────────────────────────────
# Primary backend: RF-DETR nano (Apache 2.0) on torch (OBJECT_DETECTOR_BACKEND).
# Weights downloaded by setup_assets.py. MediaPipe EfficientDet is the fallback.
rfdetr>=1.3.0

# ── Vision — place recognition image encoder ─────────────────────────────────
# MobileCLIP-S2 via open_clip (perception/place_embedder.py). Produces 512-d image
# embeddings for perception/place_recognition.py. torchvision is pinned <0.28 so it
# stays paired with torch 2.12 (torchvision 0.28 would drag in torch 2.13). Weights
# (~0.4 GB) are fetched by setup_assets.py into assets/models/mobileclip (gitignored).
# Apple's MobileCLIP pretrained weights are non-commercial, consistent with the license.
open_clip_torch>=2.24.0
torchvision>=0.18.0,<0.28

# ── Optional macOS GUI dashboard ──────────────────────────────────────────────
PySide6>=6.7.0

# ── Hardware — serial (Maestro servo controller + Arduino) ───────────────────
pyserial>=3.5

# ── macOS menu bar battery meter (tools/rex_battery_menubar.py) ───────────────
rumps>=0.4.0

# ── Fuzzy string matching (command parser + DJ vibe matching) ─────────────────
rapidfuzz>=3.9.0

# ── Audio metadata — ID3 tag reading for DJ music index ───────────────────────
mutagen>=1.47.0

# ── Config & environment ──────────────────────────────────────────────────────
python-dotenv>=1.0.1

# ── System monitoring (interoception — CPU load) ──────────────────────────────
psutil>=6.0.0

# ── HTTP (weather API, misc) ──────────────────────────────────────────────────
requests>=2.32.0

# ── ReSpeaker Flex XVF3800 parameter access (tools/flex_ctl.py; USB control
# transfers via the bundled libusb — no Homebrew libusb needed) ───────────────
pyusb>=1.2.1
libusb-package>=1.0.26
